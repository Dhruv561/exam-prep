\topic{PREREQUISITES}
\hd{Confidence intervals:} \bh{Pivot:} function of data and $\theta$ with fully known distribution.

\hd{Tests, estimators:} $H_1$: effect. \bh{Test statistic:} known distribution under $H_0$. Statistical $\ne$ practical significance. \bh{Unbiased:} Bias 0. $(n-1)S^2/\sigma^2\sim\chi^2_{n-1}$ for Normal data.
\sk\hd{Model terms:} Test a factor by one LRT on $L-1$ df, not its separate $z$ values. Observational data show \bh{association, not causation}. A predictor that flips sign is “not significant in the presence of the others”.
\sk\bh{Bivariate Normal pdf:} $f(x)=\frac{1}{2\pi|\Sigma|^{1/2}}\exp\{-\frac12(x-\mu)\T\Sigma\inv(x-\mu)\}$.
\sk\bh{More cutoffs:} $\chi^2_{df}(.95)$: 8: 15.51, 25: 37.65. $\chi^2_{df}(.99)$: 2: 9.21, 3: 11.34. $\Phi(1)=.84$, $\Phi(2)=.98$. $t_{60}(.975)=2.00$.

% ---------- overdispersion ----------
\topic{OVERDISPERSION \& QUASI-LIKELIHOOD}
\hd{Quasi-score in full:} \m{u_i=\frac{Y_i-\mu_i}{\phi v(\mu_i)}}: $\E u=0$, $\Var u=\frac1{\phi v}$, $-\E\frac{\partial u}{\partial\mu}=\frac1{\phi v}=\E u^2$, same as $\partial\log L/\partial\mu$.
% ---------- R output ----------
\topic{READING R OUTPUT}
\li\bh{Deviance Residuals} (old R): want median near 0, roughly symmetric.
\li\bh{Std. Error:} $\sqrt{V_{jj}}=\sqrt{[I(\hb)\inv]_{jj}}$.
\li\bh{Interaction tested twice:} $z$ from Fisher information $\sim\Nor(0,1)$, and $\Delta D\sim\chi^2_1$. Prefer $\Delta D$.
% ---------- EM ----------
\topic{MIXTURES \& EM}
\li Exp$(\lambda)$, Exp$(2\lambda)$: $P_{i1}=\pi/[\pi+2(1-\pi)e^{-\lambda x_i}]$.
\li $\Nor(\mu,1)$, $\Nor(2\mu,4)$: $\hmu=\frac1n\sum(P_{i1}+\frac12P_{i2})x_i$.
\li $K=3$: $N_k/\pi_k=c$ for all $k$, sum gives $c=n$.
\li\bh{Is it a maximum?} $Q''_\mu=-N_k/\sigma_k^2$; $Q''_\pi=-N_1/\pi^2-N_2/(1-\pi)^2$; $Q''_{\sigma^2}=-N_k/2\hat\sigma_k^4$.
\li\bh{Overlap:} $\pik$ less decisive, EM slower and more start-sensitive. Do not compare log-lik across different data.
\li`length(log_liks)` $=$ iterations $+1$.
% ---------- Bayes ----------
\topic{BAYES}
\li\bh{In $\lambda$:} Poisson likelihood is $\lambda^{\sum y_i}e^{-n\lambda}$, Exp is $\lambda^ne^{-\lambda\sum y_i}$: Gamma kernels. Exp(rate $b$) = Gamma$(1,b)$.
\li\bh{Exp-Gamma mean:} weight $\frac b{b+\sum x_i}$ on $a/b$, rest on $n/\sum x_i$.
\li\bh{Predictive $t$:} known mean $\mu_0$, $\tau|x\sim\Ga(a^*,b^*)$: 3-parameter $t$, $\nu=2a^*$, location $\mu_0$, scale$^2$ $s^2=b^*/a^*$, pdf $\propto(1+\frac{(x-\mu_0)^2}{\nu s^2})^{-(\nu+1)/2}$.
\li\bh{Spike prior:} $p(x|\gamma)=\int p(x|\mu)p(\mu|\gamma)d\mu$.
\li\bh{Conjugate exp family:} $f=\alpha(x)e^{x\theta+\beta(\theta)}$, prior $\propto e^{\gamma\theta+\delta\beta(\theta)}$: posterior $(\gamma+x,\delta+1)$.
\li\bh{Joint prior} $\mu|\tau$, $\tau$: exact posterior (VI box), so sample $\tau$ then $\mu$, no MCMC.
% ---------- simulation ----------
\topic{SIMULATION}
\bh{Why rejection works:} $(X,Y)$ is uniform under $kh$ (density $1/k$); points kept are uniform under $f$, so $X\sim f$.
% ---------- MCMC ----------
\topic{MCMC}
\bh{Why it works (not examinable):} detailed balance $\pi(\theta)p(\theta\to\theta')=\pi(\theta')p(\theta'\to\theta)$ makes $\pi$ stationary; aperiodic, irreducible: limiting; ergodic: averages converge.
\sk\bh{MH derivation, $x\sim\Nor(\mu,1)$, $\mu\sim\Nor(0,1)$, random walk:} $\log\frac{\pi(\mu')}{\pi(\mu)}=-\frac12[(x-\mu')^2+\mu'^2]+\frac12[(x-\mu)^2+\mu^2]$; $A=\min(1,e^{\text{this}})$.
\sk\bh{Gibbs $\mu$ derivation:} $p(\mu|\tau,x)\propto\exp\{-\frac\tau2\sum(x_i-\mu)^2-\frac\kappa2(\mu-m)^2\}\propto\exp\{-\frac12(n\tau+\kappa)\mu^2+(\tau n\bar x+\kappa m)\mu\}$: $A=n\tau+\kappa$, $B=\tau n\bar x+\kappa m$.
\sk\bh{Exp change point:} $\lambda_1|\cdot\sim\Ga(a_1+M,b_1+S_M)$, $\lambda_2|\cdot\sim\Ga(a_2+n-M,b_2+S_n-S_M)$, $S_m=\sum_1^my_i$; $p(m|\cdot)\propto(\lambda_1/\lambda_2)^me^{-(\lambda_1-\lambda_2)S_m}$.
\sk\bh{Discrete $\mu\in\{0,1\}$:} $P(\mu=1|\theta)=f(\theta,1)/[f(\theta,0)+f(\theta,1)]$.
% ---------- VI ----------
\topic{VARIATIONAL INFERENCE}
\bh{CAVI $\tau$ derivation:} $\log q_\tau^*=\E_\mu[\log p(x|\mu,\tau)+\log p(\mu|\tau)+\log p(\tau)]+c$
$=(\frac n2+\frac12+a_0-1)\log\tau-\tau[b_0+\frac12\sum\E(x_i-\mu)^2+\frac{\lambda_0}2\E(\mu-\mu_0)^2]+c$: power of $\log\tau$ is $a^*-1$, coefficient of $-\tau$ is $b^*$.
\sk\bh{ELBO, Normal-Gamma:} $\E[\log p(x,\mu,\tau)]-\E\log q_\mu-\E\log q_\tau$, with $\E\log q_\mu=-\frac12\log\sigma^{2*}+c$, $\E\log q_\tau=\log b^*-\log\Gamma(a^*)+(a^*-1)\psi(a^*)-a^*$.
\sk\bh{ELBO, bivariate target,} up to a constant, $q_i=\Nor(m_i,s_i^2)$: $-\frac12[\Lambda_{11}(s_1^2+(m_1-\mu_1)^2)+\Lambda_{22}(s_2^2+(m_2-\mu_2)^2)+2\Lambda_{12}(m_1-\mu_1)(m_2-\mu_2)]+\frac12\log s_1^2+\frac12\log s_2^2$.
\sk\bh{Bivariate prior} (Gibbs row): $q_1^*=\Nor(B/A,1/A)$ with $\mu_2$ replaced by $\E\mu_2$.
\sk\bh{Variance form:} $\sigma^2\sim\IG(a_0,b_0)$, $\mu|\sigma^2\sim\Nor(\mu_0,\sigma^2/\lambda_0)$.
