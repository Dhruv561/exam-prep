\begin{tb}{cLik}{1 \textbullet\ MLE $\to$ Fisher information $\to$ asymptotic distribution (Q1, 7--10 marks)}
\begin{steps}
\item $\ell(\theta)=\sum_i\log f(x_i;\theta)$; drop $\theta$-free terms ($+c$); collect $\sum x_i,\sum x_i^2,\sum\log x_i$. State support.
\item Score $\ell'(\theta)=0\Rightarrow\hat\theta$; check $\ell''(\hat\theta)<0$.
\item $J(\theta)=-\ell''(\theta)$ (observed). $\I(\theta)=\E[J(\theta;X)]$, \textbf{whole sample} (contains $n$): replace $\sum X_i\to n\E X$, $\sum X_i^2\to n\E X^2$. If $\ell''$ has no data, write ``$J=\I$, no expectation needed''.
\item $\hat\theta\ \dot\sim\ N\big(\theta,\I(\theta)^{-1}\big)$; est.\ var $1/\I(\hat\theta)$; Wald CI $\hat\theta\pm1.96/\sqrt{\I(\hat\theta)}$.
\end{steps}
\h{Reparametrise:} MLE invariant ($\widehat{g(\theta)}=g(\hat\theta)$); information not: redo, or $\I_\eta=\I_\theta(d\theta/d\eta)^2$. CI for $g(\theta)$: map endpoints (monotone $g$); better built on an unbounded scale ($\log$, logit) then back-transformed.
\h{Vector $\theta$:} $\I=-\E[\partial^2\ell/\partial\theta\partial\theta\tsp]$, $\V\hat\theta\approx\I^{-1}$; $\begin{psmallmatrix}a&b\\b&d\end{psmallmatrix}^{-1}\!=\!\frac{1}{ad-b^2}\begin{psmallmatrix}d&-b\\-b&a\end{psmallmatrix}$. LR region $\{\theta:2[\ell(\hat\theta)-\ell(\theta)]\le\chi^2_k(.95)\}$, $k=\dim\theta$.\\[0.3pt]
\begin{tabularx}{\linewidth}{@{}L|c|c@{}}
\textbf{$n$ iid model} & \textbf{MLE} & \textbf{$\I$ (whole sample)}\\\hline
Bernoulli$(p)$ & $\bar x$ & $n/[p(1-p)]$\\
$Y_i\sim$Bin$(m_i,p)$ & $\sum y_i/\sum m_i$ & $\sum m_i/[p(1-p)]$\\
\quad same, in logit $\eta=\log\frac p{1-p}$ & $\log\frac{\hat p}{1-\hat p}$ & $\sum m_i\,p(1-p)$\\
Poisson$(\lambda)$ & $\bar x$ & $n/\lambda$\\
Exp(rate $\lambda$) & $1/\bar x$ & $n/\lambda^2$\\
Exp(mean $\theta$), $f=\theta^{-1}e^{-x/\theta}$ & $\bar x$ & $n/\theta^2$\\
$N(\mu,\sigma^2)$, $\sigma^2$ known & $\bar x$ & $n/\sigma^2$\\
$N(\mu_0,1/\tau)$, $\mu_0$ known, precision & $n/\sum(x_i-\mu_0)^2$ & $n/(2\tau^2)$\\
$N(\mu_0,v)$, $\mu_0$ known, variance & $\sum(x_i-\mu_0)^2/n$ & $n/(2v^2)$\\
Gamma(shape $a$ known, rate $b$) & $a/\bar x$ & $na/b^2$\\
Geometric $p(1-p)^x$, $x\ge0$ & $1/(1+\bar x)$ & $n/[p^2(1-p)]$\\
$\theta x^{\theta-1}$ on $(0,1)$ & $-n/\sum\log x_i$ & $n/\theta^2$\\
\end{tabularx}
\h{Moments:} Exp(rate $\lambda$) $\E X^2=2/\lambda^2$ \sep Gamma$(a,b)$ $\E X^2=\frac{a(a+1)}{b^2}$, $\E\frac1X=\frac b{a-1}$ \sep Poisson $\E X^2=\lambda+\lambda^2$ \sep $N$: $\E(X-\mu)^2=\sigma^2$, $\E(X-\mu)^4=3\sigma^4$ \sep $\theta x^{\theta-1}$: $\E[-\log X]=\frac1\theta$ \sep Pareto $\alpha x_m^\alpha x^{-\alpha-1}$: $\E\log\frac X{x_m}=\frac1\alpha$ \sep lognormal $\E Y^k=e^{km+k^2\sigma^2/2}$.\\
\trap course $\I$ already has $n$ (don't divide again) \sep use the parameter named (rate/mean, precision/variance) \sep $\ell''$ with data needs $\E$.
\end{tb}

\begin{tb}{cLik}{2 \textbullet\ Binomial regression (logit) and CIs}
$Y_i\sim$Bin$(m_i,p_i)$ indep., $g(p_i)=\eta_i=x_i\tsp\beta$ (name dist., independence, link, $\eta$).
$\ell(\beta)=c+\sum_i[y_i\eta_i-m_i\log(1+e^{\eta_i})]$; score $\sum_i(y_i-m_ip_i)x_i$; $\I(\beta)=J(\beta)=X\tsp WX$, $W=\text{diag}(m_ip_i(1-p_i))$ (canonical: no $y$ in Hessian; probit/cloglog $J\ne\I$).\\
\h{Links} $\eta=g(p)\leftrightarrow p=g^{-1}(\eta)$: logit $\log\frac p{1-p}\leftrightarrow\frac{1}{1+e^{-\eta}}$ ($p=\frac12$ at $x=-\beta_0/\beta_1$, slope there $\beta_1/4$) \sep probit $\Phi^{-1}(p)\leftrightarrow\Phi(\eta)$ \sep cloglog $\log(-\log(1-p))\leftrightarrow1-\exp(-e^\eta)$ (asymmetric; ``$\ge1$ Poisson event'').\\
\h{CI for $p$ (or $\mu$) at $x_0$:} (1) $\hat\eta_0=x_0\tsp\hat\beta$ (+offset); (2) se$=\sqrt{x_0\tsp Vx_0}$, $V$=\rc{vcov}: $V_{00}+2x_0V_{01}+x_0^2V_{11}$ (all variances $+\,2\times$ every covariance); (3) $g^{-1}(\hat\eta_0\pm1.96\,\text{se})$. Never $\hat p\pm1.96\,\text{se}(\hat p)$; asymmetric is correct.\\
\h{Odds ratio} for $+\Delta$ in $x_j$: $e^{\hat\beta_j\Delta}$, CI $\exp\{\Delta(\hat\beta_j\pm1.96\,\text{se}_j)\}$; \%change $=(OR-1)100$. Two non-baseline levels: $\V(\hat\beta_a-\hat\beta_b)=V_{aa}+V_{bb}-2V_{ab}$. $e^\beta$ = OR only for logit (log link: rate ratio). Case--control: logit only shifts intercept $\Rightarrow$ ORs valid, $\hat p$ not population risks.\\
\say{CI on $\eta$ then transform: $\hat\eta_0$ is linear in $\hat\beta$ so approx.\ normal; $g^{-1}$ monotone keeps 95\% coverage and $p$ in $(0,1)$.}
\say{Offset: expected count $\propto$ exposure, $\mu_i=t_i\lambda_i$, so $\log t_i$ enters with coefficient 1; $e^{\beta_j}$ = rate ratio per unit exposure.}
\end{tb}

\begin{tb}{cLik}{3 \textbullet\ Deviance, LRT, AIC (every GLM)}
$D=2[\ell_{\rm sat}-\ell(\hat\beta)]$ on $n-p$ df; scaled $D/\phi$ ($=D$ binomial/Poisson); $0\log0=0$.
Binomial $d_i=2[y_i\log\frac{y_i}{\hat\mu_i}+(m_i-y_i)\log\frac{m_i-y_i}{m_i-\hat\mu_i}]$ \sep Poisson $d_i=2[y_i\log\frac{y_i}{\hat\mu_i}-(y_i-\hat\mu_i)]$ ($y_i{=}0$: $2\hat\mu_i$) \sep Normal $(y_i-\hat\mu_i)^2$ \sep Gamma $-2[\log\frac{y_i}{\hat\mu_i}-\frac{y_i-\hat\mu_i}{\hat\mu_i}]$.\\
\h{Counting:} $n=$ null df$+1$; $p=n-$resid df (incl.\ intercept; $k$-level factor $=k-1$; offset not counted).\\
\h{GOF} ($\phi=1$ only): $D\approx\chi^2_{n-p}$ if all $m_i\hat p_i,m_i(1-\hat p_i)\ge5$ (Poisson $\hat\mu_i\ge5$). Invalid for 0/1 or sparse data, or $\phi$ estimated. $\E\chi^2_k=k$, so $D/\text{df}\approx1$ is reassuring.\\
\h{LRT} A$\subset$B: $D_A-D_B\approx\chi^2_s$, $s=\text{df}_A-\text{df}_B$; fine for 0/1 data. Factor = one block on $k-1$ df (dummy $z$s don't answer ``does it matter''). Wald $z=\hat\beta/\text{se}$, $z^2\approx$LR; disagree $\Rightarrow$ trust LRT. Different links/non-nested: AIC only.\\
\h{AIC}$=2p-2\hat\ell$, smaller better; $\text{AIC}_B-\text{AIC}_A=2s+D_B-D_A$ (extra term worth it iff $\Delta D>2$ per param).
\[\hat\ell=p-\tfrac{\text{AIC}}2,\qquad \ell_{\rm sat}=\hat\ell+\tfrac D2\]
($\ell_{\rm sat}$ is the same for every model on the data: a check.) 
Gamma/normal: R counts $\phi$, $\hat\ell=(p+1)-\text{AIC}/2$. Quasi: AIC NA (use Poisson fit).\\
\say{Adequate: $D=\_$ on $n-p$ df $<\chi^2_{n-p}(.95)=\_$; valid as all fitted counts $\ge5$, so no evidence of lack of fit. (Sparse/0-1 data: test not valid, say so.)}\\
\say{Prefer B: LRT $D_A-D_B=\_$ on $s$ df $>\chi^2_s(.95)=\_$ (p$=\_$), reject $H_0$: extra coefs $=0$; and AIC$_B<$AIC$_A$.} (``two quantitative reasons'')
\end{tb}

\begin{tb}{cGLM}{6 \textbullet\ Exponential family, canonical link, $v(\mu)$ (8--10 marks)}
\[\begin{gathered}f(y;\theta,\phi)=\exp\Big[\frac{y\theta-b(\theta)}{a(\phi)}+c(y,\phi)\Big],\quad a(\phi)=\phi\\ \E Y=b'(\theta),\qquad \V Y=a(\phi)b''(\theta)=a(\phi)v(\mu)\end{gathered}\]
\begin{steps}
\item Take logs; \textbf{say support doesn't depend on the parameter} (Unif$(0,\theta)$, shifted exp: not EF).
\item Term $y\times h(\text{param})$: $\theta=h$ (param must multiply $y$, not $y^2,\log y,e^y$).
\item Factor scale out of $y$-term \& param-only term: $a(\phi)$.
\item Param-only term $=-b/a(\phi)$: \textbf{rewrite $b$ in $\theta$} (invert $\theta(\mu)$). Rest $=c(y,\phi)$ (known consts OK).
\item Check $b'(\theta)=\mu$. $v(\mu)=b''(\theta)$ at $\theta=\theta(\mu)$; $\V Y=a(\phi)v(\mu)$ ($\ne v$ unless $\phi=1$).
\item Canonical link $g=(b')^{-1}$: solve $\mu=b'(\theta)$, $g(\mu)=\theta(\mu)$.
\end{steps}
\begin{tabularx}{\linewidth}{@{}L|c|c|c|c@{}}
dist & $\theta(\mu)=$ canon.\ $g$ & $b(\theta)$ & $\phi$ & $v(\mu)$\\\hline
$N(\mu,\sigma^2)$ & $\mu$ & $\theta^2/2$ & $\sigma^2$ & 1\\
Poisson & $\log\mu$ & $e^\theta$ & 1 & $\mu$\\
Bernoulli & $\log\frac{\mu}{1-\mu}$ & $\log(1+e^\theta)$ & 1 & $\mu(1-\mu)$\\
Bin$(m)$ count & $\log\frac{\mu}{m-\mu}$ & $m\log(1+e^\theta)$ & 1 & $\mu(1-\frac\mu m)$\\
Bin prop.\ $Y/m$ & logit $\mu$ & $\log(1+e^\theta)$ & $1/m$ & $\mu(1-\mu)$\\
Exp(rate $\lambda$) & $-1/\mu$ & $-\log(-\theta)$ & 1 & $\mu^2$\\
Gamma$(\mu,\nu)$ & $-1/\mu$ & $-\log(-\theta)$ & $1/\nu$ & $\mu^2$\\
Inv.\ Gauss. & $-1/(2\mu^2)$ & $-\sqrt{-2\theta}$ & $1/\lambda$ & $\mu^3$\\
Geom.\ $p(1{-}p)^y$ & $\log\frac{\mu}{1+\mu}$ & $-\log(1-e^\theta)$ & 1 & $\mu(1+\mu)$\\
NegBin, $r$ known & $\log\frac{\mu}{\mu+r}$ & $-r\log(1-e^\theta)$ & 1 & $\mu+\frac{\mu^2}r$\\
mean of $k$ Pois.\ & $\log\mu$ & $e^\theta$ & $1/k$ & $\mu$\\
\end{tabularx}
Natural $\theta$: Exp $-\lambda$; Geom $\log(1-p)$; Bernoulli logit $p$. Gamma$(\mu,\nu)$ = mean, shape; R's ``inverse'' link is $1/\mu$. Gamma: $\log f=\frac{y(-1/\mu)-\log\mu}{1/\nu}+\nu\log\nu+(\nu-1)\log y-\log\Gamma(\nu)$. Lemma proof: $\E\ell'=0$, $\E\ell''=-\E(\ell')^2$ (one obs).\\
\h{Canonical link $\Rightarrow$} $\ell=\sum[y_ix_i\tsp\beta-b(x_i\tsp\beta)]/\phi+c$: $X\tsp y$ sufficient; score $X\tsp(y-\mu)/a(\phi)$; Hessian has no $y$ so $J=\I$, Newton--Raphson = Fisher scoring. Log link often preferred over $-1/\mu$, $\log\frac{\mu}{\mu+r}$: any $\eta\in\mathbb R$ valid, multiplicative effects.
\end{tb}

\begin{tb}{cGLM}{7 \textbullet\ GLM fitting (IWLS), unknown $\phi$, diagnostics}
GLM: indep.\ $Y_i$ from one EF (common $\phi$), $\eta_i=x_i\tsp\beta$, monotone differentiable $g(\mu_i)=\eta_i$; $\phi$ not modelled.
\[\begin{gathered}U_j=\sum_i\frac{(y_i-\mu_i)x_{ij}}{a(\phi)v(\mu_i)g'(\mu_i)},\qquad W_{ii}=\frac1{g'(\mu_i)^2v(\mu_i)}\\ z_i=\eta_i+(y_i-\mu_i)g'(\mu_i),\qquad \hat\beta^{\rm new}=(X\tsp WX)^{-1}X\tsp Wz\end{gathered}\]
Start $\hat\mu=y$; iterate (= Fisher scoring). $\V\hat\beta=a(\phi)(X\tsp\hat WX)^{-1}$: $\phi$ cancels in update, not variance $\Rightarrow$ SE$\,\propto\sqrt{\hat\phi}$. Canonical $W_{ii}=v(\mu_i)$ (Poisson-log $\mu_i$, logit $m_ip_i(1-p_i)$); Gamma-log $W=1$. $W^{-1}\propto\V z_i$ (delta method).\\
\h{$\phi$ unknown} (Gamma $\V Y=\phi\mu^2$, CV$=\sqrt\phi$; normal): $\hat\phi=X^2/(n-p)$, $X^2=\sum(y_i-\hat\mu_i)^2/v(\hat\mu_i)$ (Pearson, \emph{not} $D/(n-p)$); $t$ tests, F tests, no deviance GOF ($\hat\phi$ uses same residuals).\\
\h{Residuals:} Pearson $r_P=\frac{y-\hat\mu}{\sqrt{v(\hat\mu)}}$, $\sum r_P^2=X^2$ \sep deviance $r_D=\text{sign}(y-\hat\mu)\sqrt{d_i}$, $\sum r_D^2=D$ (R default) \sep standardised $r/\sqrt{(1-h_{ii})\hat\phi}$ \sep jack-knife \rc{rstudent}: large $\Rightarrow$ outlier. \h{Leverage} $h_{ii}$: diag of IWLS hat matrix, $\sum h_{ii}=p$, flag $>2p/n$ (potential only). \h{Cook's} $C_i\approx\frac{r_P^2h_{ii}}{p\hat\phi(1-h_{ii})^2}$: needs leverage \emph{and} residual; refit without $i$ to confirm. \h{Plots} $r_D$ vs $\hat\eta$: trend $\Rightarrow$ wrong mean/link; changing spread $\Rightarrow$ wrong $v(\mu)$; bands = discrete $y$. Half-normal: tail jump.
\end{tb}

\begin{tb}{cQ}{8 \textbullet\ Overdispersion and quasi-likelihood}
\h{Def:} actual $\V Y_i>$ model's (Poisson $\mu_i$; binomial $m_ip_i(1-p_i)$). Causes: dependent trials/events (clustering); $p$ or rate varying between units (missing covariates). Large $D$ can also be wrong mean structure/outliers: check residuals vs $\hat\eta$ first. 0/1 data can't show it.\\
\h{Quantitative reasons:} (i) resid $D\gg$ resid df; (ii) GOF \rc{pchisq(D,df,lower=F)} tiny (counts $\ge5$); (iii) $\hat\phi=X^2/(n-p)\gg1$.\\
\say{Ignoring OD: $\hat\beta$ unaffected ($\phi$ cancels from score) but true $\V\hat\beta$ is $\phi\times$ reported: SEs too small, $z$ too big, p-values too small, CIs too narrow, $\chi^2$ LRTs too significant $\Rightarrow$ predictors wrongly declared significant/kept.}\\
\h{Quasi-likelihood:} specify only $g(\mu_i)=x_i\tsp\beta$, $\E Y_i=\mu_i$, $\V Y_i=\phi v(\mu_i)$ (IWLS needs only $g,v$). $u_i=\frac{Y_i-\mu_i}{\phi v(\mu_i)}$: $\E u_i=0$, $\V u_i=\frac1{\phi v}=-\E\frac{\partial u_i}{\partial\mu_i}$ (acts like a score).
\[\begin{gathered}Q_i=\int_{y_i}^{\mu_i}\frac{y_i-t}{\phi v(t)}dt\le0,\qquad Q=\sum_i Q_i\\ \frac{D_Q}{\phi}=-2\sum_i Q_i\ \ (Q_{\rm sat}=0),\qquad D_Q=-2\sum_i\int_{y_i}^{\mu_i}\frac{y_i-t}{v(t)}dt\end{gathered}\]
\begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}p{0.2\linewidth}|L|>{\raggedright\arraybackslash}p{0.14\linewidth}@{}}
$v(\mu)$ & $\phi Q_i$ (definite integral; slides drop $\mu$-free terms) & $D_Q=$\\\hline
$\mu$ (q-Poisson) & $y\log\frac\mu y-(\mu-y)$ \ [slides: $\frac1\phi(y\log\mu-\mu)$] & Poisson dev.\\
$\frac{\mu(m-\mu)}m$ (q-binomial) & $y\log\frac\mu y+(m-y)\log\frac{m-\mu}{m-y}$ & binomial dev.\\
$\mu^2$ & $-\frac y\mu+1-\log\frac\mu y$ & Gamma dev.\\
\end{tabularx}

\h{Quasi-Poisson/binomial output:} same $\hat\beta$, fitted values, predictions, deviances; SE$_q=\sqrt{\hat\phi}\,$SE$_{\phi=1}$; $t=z/\sqrt{\hat\phi}$; AIC NA; no GOF; compare nested models by

\begin{tb}{cEM}{9 \textbullet\ Mixtures and EM (10--12 marks): $Q$, E-step, M-step}
$Z_i\sim$Cat$(\pi_1..\pi_K)$ ($K{=}2$: $\pi,1-\pi$), $X_i|Z_i{=}k\sim f(\cdot;\theta_k)$; pick component then sample (not $\pi_1X_1+\pi_2X_2$).
\h{Incomplete} $\ell(\theta)=\sum_i\log\sum_k\pi_kf(x_i;\theta_k)$: log of a sum, no closed form.
\[\begin{aligned}
P(X,Z\mid\theta)&=\textstyle\prod_{i}\prod_{k}[\pi_kf(x_i;\theta_k)]^{I(Z_i=k)}\\
\log P(X,Z\mid\theta)&=\textstyle\sum_i\sum_kI(Z_i{=}k)[\log\pi_k+\log f(x_i;\theta_k)]\\

&=\textstyle\sum_i\sum_kE[I(Z_i{=}k)|X,\theta^0][\,\cdots\,]\\
&=\textstyle\sum_i\sum_k\gamma_{ik}[\log\pi_k+\log f(x_i;\theta_k)]
\end{aligned}\]
(line 2 = complete log-lik; line 4 by linearity; line 5 since $E[I(Z_i{=}k)|X,\theta^0]=P(Z_i{=}k|X_i,\theta^0)$.)

\begin{tb}{cBay}{12 \textbullet\ Metropolis--Hastings (the $\alpha$ ledger)}
Propose $\theta^*\sim q(\cdot|\theta^c)$; $u\sim U(0,1)$; accept if $u<\alpha$, else repeat $\theta^c$. $\pi=p(x|\theta)p(\theta)$ unnormalised ($p(x)$ cancels).
\[\alpha=\min\Big\{1,\ \frac{\pi(\theta^*)}{\pi(\theta^c)}\cdot\frac{q(\theta^c|\theta^*)}{q(\theta^*|\theta^c)}\Big\}\quad\text{(reverse/forward)}\]
\h{Ledger} (one line per factor, 4 dp, round once): (i) likelihood ratio incl.\ $(\tau^*/\tau^c)^{n/2}$; (ii) each prior ratio (keep $\sqrt\tau$ of $p(\mu|\tau)$); (iii) proposal ratio per proposed parameter; multiply; $\min\{1,\cdot\}$. Logs: accept iff $\log u<\log\pi^*-\log\pi^c+\log q(c|*)-\log q(*|c)$.
\h{Reverse density} starts at $\theta^*$, lands on $\theta^c$: a rate/variance set by $\theta^c$ going forward is set by $\theta^*$ in reverse. E.g.\ Exp(rate $1/\tau^c$): ratio $=\frac{(1/\tau^*)e^{-\tau^c/\tau^*}}{(1/\tau^c)e^{-\tau^*/\tau^c}}$.
Symmetric RW ($N(\theta^c,s^2)$, $U[\theta^c\pm\epsilon]$): ratio 1, $\alpha=\min\{1,\frac{\pi^*}{\pi^c}\}$ \sep independence $g$: $\alpha=\min\{1,\frac{\pi^*/g^*}{\pi^c/g^c}\}$ \sep $\log\tau^*\sim N(\log\tau^c,s^2)$: extra $\tau^*/\tau^c$ \sep Gibbs: $\alpha=1$.\\
\say{Stationary distribution = the target $\pi$ (normalised); with symmetric $q$, read $\pi$ off $\alpha=\min\{1,\pi^*/\pi^c\}$.} \say{Acceptance too high (tiny steps) or too low (stuck, flat trace) both mix slowly; aim $\approx0.2$--$0.5$.}\\
\trap no proposal ratio for asymmetric $q$ ($\approx$2 marks) \sep forward/reverse swapped \sep dropping $\tau^{n/2}$ \sep rounding mid-ledger.
\end{tb}

\begin{tb}{cMC}{13 \textbullet\ Monte Carlo, inversion, rejection}
MC: $\E g(\theta)\approx\frac1N\sum g(\theta^{(s)})$; $P(A)\approx$ mean of indicators; SE $=\text{sd}(g)/\sqrt N$ or $\sqrt{\hat p(1-\hat p)/N}$ (independent draws; not the posterior sd).
Inversion: $F^{-1}(U)\sim F$; solve $u=F(x)$; Exp(rate $\lambda$): $x=-\log(1-u)/\lambda$. Discrete: \rc{x<-0; while(F(x)<U) x<-x+1}.\\
\h{Rejection} (target $\tilde f\propto f$, $Z=\int\tilde f$; proposal $g$): $X\sim g$, $U\sim U(0,1)$; accept if $U\le\frac{\tilde f(X)}{Mg(X)}$ [slides: $Y\sim U(0,kh(X))$, accept $Y<f(X)$]. Smallest $M=\sup\tilde f/g$: set $\frac{d}{dx}\log\frac{\tilde f}{g}=0$, check max/endpoints. $P(\text{acc})=Z/M$ ($1/M$ if normalised); tries $\sim$Geom, mean $M/Z$. Uniform on $[a,b]$: $M=(b-a)\max\tilde f$. Gamma$(m,\lambda)$ via Exp$(\mu)$: best $\mu=\lambda/m$, $k^*=\frac{m^me^{-(m-1)}}{\Gamma(m)}$ ($m>1$). Prior as proposal: accept if $U\le p(x|\theta)/p(x|\hat\theta_{\rm MLE})$.\\
\say{Rejection needs only $\tilde f$ and gives independent exact draws, but needs an envelope; in high dimensions $M$ is huge and nearly all draws are rejected $\Rightarrow$ MCMC.}
\end{tb}

\begin{tb}{cMC}{14 \textbullet\ MCMC diagnostics}
Trace: converged = ``hairy caterpillar'', no trend, constant spread; chains from \emph{overdispersed} starts overlap. Burn-in = discard iterations before chains overlap and stabilise.
$\hat R=B/W$: $B$ = sd of pooled post-burn-in draws (all chains), $W$ = mean within-chain sd; $\hat R<1.05$ accept; $\hat R\gg1$ not converged ($\to1$ as $T\to\infty$): run longer / better proposal.
Lag-$k$ autocorrelation $\rho_k$ (ACF). ESS$=\frac{n}{1+2\sum_{i\ge1}\rho_i}\approx n\frac{1-\rho_1}{1+\rho_1}$; MCMC SE $=s/\sqrt{\text{ESS}}$ (not $s/\sqrt n$); aim ESS$>4000$. Thinning (every $k$th): saves memory, loses information, never raises ESS.\\
\say{Burn-in from trace: discard up to the iteration where the chains from different starts merge and stop trending (e.g.\ first $\_$ of $\_$); use the rest.}
\say{Run several chains from widely separated starts. Informal: trace plots overlap, stationary, no trend; ACF decays fast. Formal: BGR $\hat R=B/W<1.05$ on post-burn-in draws. Passing doesn't prove convergence: all chains may miss a region.}
\end{tb}

\begin{tb}{cMC}{15 \textbullet\ Variational inference (CAVI)}
$q^*=\arg\min_{q\in\mathcal D}\text{KL}(q\|p(\theta|x))$, KL$=E_q[\log q-\log p(\theta|x)]\ge0$ (not symmetric). ELBO$(q)=E_q[\log p(\theta,x)]-E_q[\log q]$; $\log p(x)=\text{ELBO}+\text{KL}$, so ELBO$\le\log p(x)$; max ELBO $\iff$ min KL, no $p(x)$ needed. Mean field $q=\prod_jq_j(\theta_j)$ (assumption on the approximation, not the model).\\
\textbf{CAVI:} $q_j^*(\theta_j)\propto\exp\{E_{-j}[\log p(\theta,x)]\}$; ELBO never decreases; local optimum; stop when $\Delta$ELBO small; several starts, keep highest ELBO. Recipe: log joint line by line, keep terms with $\theta_j$, take $E_{-j}$, expand, name. $E_\mu[(c-\mu)^2]=(c-E\mu)^2+V\mu$; $E\tau=a^*/b^*$.\\
\h{Joint prior} $\mu|\tau\sim N(m_0,\frac1{\kappa_0\tau})$, $\tau\sim$Ga$(a,b)$ (lecture):
\[\begin{aligned}q^*_\mu&=N\Big(\frac{\kappa_0m_0+n\bar x}{\kappa_0+n},\ \frac1{(\kappa_0+n)E\tau}\Big)\\
q^*_\tau&=\text{Ga}(a^*,b^*),\qquad a^*=a+\tfrac{n+1}2\\
b^*&=b+\tfrac12\big[\textstyle\sum(x_i-E\mu)^2+nV\mu\big]+\tfrac{\kappa_0}2\big[(E\mu-m_0)^2+V\mu\big]\end{aligned}\]
\h{Independent} $\mu\sim N(m_0,\frac1{\kappa_0})$: $q^*_\mu=N\big(\frac{\kappa_0m_0+E\tau\,n\bar x}{\kappa_0+nE\tau},\frac1{\kappa_0+nE\tau}\big)$, $q^*_\tau=$Ga$\big(a+\frac n2,\ b+\frac12[\sum(x_i-E\mu)^2+nV\mu]\big)$. Iterate $q_\mu\to q_\tau\to\cdots$.
IG prior on $\sigma^2$: $q^*_{\sigma^2}$ = IG with the same $b^*$; uses $E[1/\sigma^2]=a^*/b^*$.\\
\say{VI vs MCMC: VI optimises: fast, deterministic, scales to big data/many models, but mean-field ignores posterior dependence, underestimates variance, no exactness guarantee. MCMC: slower, needs convergence checks, asymptotically exact.}
\end{tb}
